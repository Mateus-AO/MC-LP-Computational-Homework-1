\section*{Parte 2: Instância Parametrizada por Matrícula}
$$
\min_{x\geq0}-(d_1+1)x_1-(d_2+2)x_2-(d_3+1)x_3
$$
$$\text{s.a.}\quad
\begin{pmatrix}
2 & 1 & 1\\
1 & 2 & d_4+1\\
1 & 1 & 2
\end{pmatrix}
\begin{pmatrix}
x_1\\x_2\\x_3
\end{pmatrix}
\leq
\begin{pmatrix}
d_1\\d_2\\d_3
\end{pmatrix}
$$
Substituindo os valores
$
d_1=3,\,
d_2=3,\,
d_3=5,\,
d_4=0
$:

$$
\min_{x\geq0}-4x_1-5x_2-6x_3
$$
$$\text{s.a.}\quad
\begin{pmatrix}
2 & 1 & 1\\
1 & 2 & 1\\
1 & 1 & 2
\end{pmatrix}
\begin{pmatrix}
x_1\\x_2\\x_3
\end{pmatrix}
\leq
\begin{pmatrix}
3\\3\\5
\end{pmatrix}
$$
